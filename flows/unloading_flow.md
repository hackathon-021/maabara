## תהליך קבלת הציוד

```mermaid
flowchart TD
    A["כניסה למסך הראשי באפליקציה"] --> B["לחיצה על 'פינוי ציוד'"]
    B --> C["לחיצה על 'קבלת ציוד'"]
    C -.הערה.-> Cn["בחירה מתוך תפריט הכולל 4 פעולות:<br/>יצירת אריזה, יצירת הובלה, קבלת ציוד, פיזור ציוד"]
    C --> D["בחירת יחידת ההובלה מתוך רשימת<br/>יחידות ההובלה שבדרך"]
    D --> E["פתיחת חלון עם כל יחידות האריזה<br/>שהועמסו על יחידת ההובלה"]
    E --> F["בדיקה וסימון V לכלל<br/>יחידות האריזה שהגיעו"]
    F --> G["לחיצה על 'סיום פריקה'"]
    G -.הערה.-> Gn1["סטאטוס יחידת האריזה ישתנה ל'אריזה התקבלה'<br/>כ'פריט התקבל' למעט יחידת אריזה קרטון אישי"]
    G -.הערה.-> Gn2["סטאטוס יחידת ההובלה ישתנה<br/>ל'יחידת הובלה נפרקה במלואה'"]
    G --> H{"בדיקה — האם כל<br/>הארזיות התקבלו?"}
    H -->|"לא"| I["התגלה חוסר באריזה<br/>ביחידת ההובלה?"]
    I -.הערה.-> In["תתקבל הודעת אזהרה:<br/>'שים לב, לא כל הארזיות נפרקו'"]
    I --> J["הצגת רשימת כלל הארזיזות<br/>שלא נפרקו לעדכון נוסף"]
    J -.הערה.-> Jn["עדכון נוסף ליחידות האריזה<br/>לאחר בדיקה נוספת"]
    J --> K["לחיצה על סיום העדכון להמשך"]
    K -.הערה.-> Kn["סטאטוס של יחידות האריזה החסרות ישתנו ל'אריזה חסרה'<br/>כולל עדכון של סטאטוס נוסף כ'פריט חסר'"]
    K --> L["סגירה אוטומטית של יחידת ההובלה<br/>וחזרה למסך הראשי"]
    H -->|"כן"| L
    L -.הערה.-> Ln1["סטאטוס יחידת ההובלה ישתנה ל'יחידת הובלה שוחררה'<br/>ושליחת SMS עם כל פרטי ההובלה כולל חוסרים"]
    L -.הערה.-> Ln2["במידה ויתגלה עודף ביחידת אריזה, יבוצע עדכון<br/>לסטאטוס יחידת האריזה והפריט באופן ידני להתקבל"]
```
