## תהליך ההובלה 

```mermaid
flowchart TD
    W["כניסה למסך הראשי באפליקציה"] --> X["לחיצה על 'פינוי ציוד'<br/>ולאחר מכן בחירת יחידה"]
    X --> Y["לחיצה על 'יצירת הובלה'"]
    Y -.הערה.-> Yn["בחירה מתוך תפריט הכולל 4 פעולות:<br/>יצירת אריזה, יצירת הובלה, קבלת ציוד, פיזור ציוד"]
    Y --> Z["בחירת סוג יחידת ההובלה"]
    Z -.הערה.-> Zn["מתוך תפריט הכולל: משאית, אחר<br/>(אם נבחר אחר — נבחר להוסיף אופציה לפירוט)"]
    Z --> AA["פתיחת יחידת הובלה<br/>והזנת מס' רישוי"]
    AA -.הערה.-> AAn1["תאריך ההובלה ייגזר אוטומטית מהמערכת"]
    AA -.הערה.-> AAn2["סטאטוס יחידת ההובלה ישתנה<br/>'בתהליך העמסה'"]
    AA --> AB["לחיצה על 'שמירה<br/>לפתיחת יחידת ההובלה'"]
    AB --> AC["סימון ידני ובחירת יחידות האריזה<br/>הסגורות הנבחרות להובלה"]
    AC -.הערה.-> ACn["יתקבל מסך המציג את כל<br/>יחידות האריזה הסגורות"]
    AC --> AD["לחיצה על 'סיום העמסה'"]
    AD -.הערה.-> ADn1["סטאטוס יחידת האריזה<br/>ישתנה ל'אריזה בדרך'"]
    AD -.הערה.-> ADn2["סטאטוס יחידת ההובלה<br/>ישתנה 'יחידת הובלה בדרך'"]
    AD --> AE["קבלת חלון קופץ:<br/>'יחידת הובלה הועמסה'"]
    AE -.פירוט.-> AEn1["מס' רישוי יחידת ההובלה<br/>כמות יחידות הארזיה<br/>תאריך ושעה"]
    AE -.הערה.-> AEn2["שליחת SMS עם כל הפרטים<br/>של יחידת ההובלה לרשימת תפוצה"]
    AE --> AF{"האם להמשיך בהעמסה?"}
    AF -->|"לא"| AG["חזרה למסך הראשי<br/>של האפליקציה"]
    AF -->|"כן"| AH["חזרה למסך יצירת יחידת הובלה<br/>חדשה תחת אותה יחידה"]
```
