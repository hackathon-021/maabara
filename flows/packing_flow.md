## נספח א׳ — תהליך האריזה 

```mermaid
flowchart TD
    A["כניסה למסך הראשי באפליקציה"] --> B["לחיצה על 'פינוי ציוד'<br/>ולאחר מכן לחיצה בחירת יחידה"]
    B -.הערה.-> Bn["בחירת עגף, מדור וחדר ולאחר מכן לחיצה על 'יצירת אריזה'<br/>בחירה מתוך תפריט הכולל 4 פעולות:<br/>יצירת אריזה, יצירת הובלה, קבלת ציוד, פיזור ציוד"]
    B --> C["בחירה/הזנת מספר מדור<br/>מתוך תפריט המדורים (קבוצות)"]
    C --> D["בחירה/הזנת מספר חדר"]
    D --> E{"האם החדר מופה?"}
    E -->|"לא"| F["תופיע הודעת שגיאה:<br/>'יש לסיים את המיפוי'"]
    E -->|"כן"| G["בחר סוג יחידת אריזה"]
    G -.הערה.-> Gn["מתוך תפריט: קרטון מקצועי, קרטון אישי,<br/>משטח, דולב, תפזורת"]
    G --> H["לחץ על אישור<br/>לפתיחת יחידת האריזה"]
    H -.הערה.-> Hn["קרטון אישי  /  שאר יחידות האריזה"]
    H --> I["בחר את הפריטים לאריזה"]
    I -.הערה.-> In["מתוך מסך המציג את כל הפריטים בסטאטוס עובר והנצלה —<br/>בהנצלה בלבד ממוינים לפי א-ב"]
    I --> J["אשר את פעולת האריזה"]
    J -.הערה.-> Jn["בקרטון אישי אין סימון של הפריטים<br/>סטאטוס יחידת האריזה תשתנה ל'אריזה בתהליך'"]
    J --> K["הזן נתוני יעד: בניין, קומה, חדר"]
    K --> L["לסיום האריזה ביחידת האריזה,<br/>לחץ על סיום אריזה"]
    L -.הערה.-> Ln["סטאטוס יחידת האריזה תשתנה ל'אריזה נסגרה'<br/>יתקבל מספר רץ חד-חד ערכי בן 5 ספרות לזיהוי יחידת האריזה<br/>סטאטוס נוסף לפריט יהיה 'פריט נארז'"]
    L --> M(["ממשיך בעמוד הבא"])
```

## המשך תהליך האריזה

```mermaid
flowchart TD
    N["קבלת חלון קופץ:<br/>'יחידת אריזה הושלמה'"] --> O["מעבר לבדיקת תנאים אוטומטיים<br/>להמשך תהליך האריזה"]
    N -.פירוט.-> Nn["מס' יחידת האריזה בעל 5 ספרות<br/>נתוני יעד<br/>מס' החדר שממנו נארז<br/>מס' מלא של החדר<br/>שם אחראי מדור ואחראי חדר<br/>מס' מזהה של האורז"]
    O -.הערה.-> On["בקרטון אישי לא מבוצע בדיקת תנאים<br/>ובכך הסתיים תהליך האריזה"]
    O --> P{"בדיקה במסד הנתונים —<br/>האם נותרו בחדר פריטים לאריזה<br/>בסטאטוס עובר/הנצלה?"}
    P -->|"לא"| Q{"בדיקה במסד הנתונים —<br/>האם נותרו בחדר פריטים<br/>בסטאטוס גריטה?"}
    P -->|"כן"| R{"האם להמשיך באריזה?"}
    Q -->|"לא"| S["שנה את סטאטוס החדר<br/>ל'חדר סגור'"]
    Q -->|"כן"| T["שנה את סטאטוס החדר<br/>ל'ממתין לגריטה'"]
    R -->|"לא"| U["הצגת 2 בחירות:<br/>1. לא נותרו פריטים לאריזה<br/>2. הפסקה זמנית באריזה"]
    R -->|"כן"| V["מעבר לפתיחת יחידת אריזה נוספת והמשך אריזה<br/>(חזרה על תהליך האריזה באותו החדר שנבחר)"]
```
